        database.ref('products').on('value', (snapshot) => {
            const data = snapshot.val();
            products = [];
            if (data) {
                for (let key in data) {
                    let item = data[key];
                    // එක පින්තූරයක් තිබුණද, ලැයිස්තුවක් තිබුණද එය Array එකක් බවට සැකසීම
                    let imagesArray = item.images || (item.img ? [item.img] : []);
                    products.push({ firebaseId: key, ...item, images: imagesArray });
                }
            } else {
                const initialProducts = [
                    { name: "DBD Performance Graphic Tee", price: 3800, images: ["https://images.unsplash.com/photo-1521572267360-ee0c2909d518?auto=format&fit=crop&w=600&q=80"] },

                ];
                initialProducts.forEach(item => database.ref('products').push(item));
            }
            renderProducts();
        });

        function renderProducts() {
            const grid = document.getElementById('product-grid');
            if(products.length === 0) {
                grid.innerHTML = "<p>No products available.</p>";
                return;
            }
            grid.innerHTML = products.map((p, index) => {
                const mainImg = p.images.length > 0 ? p.images[0] : '';
                const thumbsHtml = p.images.length > 1 ? `
                    <div class="thumb-list">
                        ${p.images.map((imgUrl, imgIdx) => `
                            <img src="${imgUrl}" class="thumb-img ${imgIdx === 0 ? 'active' : ''}" onclick="changeMainImg(${index}, '${imgUrl}', this)">
                        `).join('')}
                    </div>
                ` : '';

                return `
                    <div class="product-card">
                        <img src="${mainImg}" class="product-img" id="main-img-${index}" alt="${p.name}">
                        ${thumbsHtml}
                        <div class="product-info">
                            <div class="product-title">${p.name}</div>
                            <div class="product-price">LKR ${Number(p.price).toLocaleString()}</div>
                            <select class="size-select" id="size-${index}">
                                <option value="Small (S)">Size: Small (S)</option>
                                <option value="Medium (M)" selected>Size: Medium (M)</option>
                                <option value="Large (L)">Size: Large (L)</option>
                                <option value="XL">Size: XL</option>
                            </select>
                            <button class="add-btn" onclick="addToCart(${index})">Add To Cart</button>
                            ${isAdminLoggedIn ? `<button class="delete-prod-btn" onclick="deleteProduct('${p.firebaseId}')">🗑️ Delete Product</button>` : ''}
                        </div>
                    </div>
                `;
            }).join('');
        }

        function changeMainImg(productIndex, imgUrl, thumbElement) {
            document.getElementById(`main-img-${productIndex}`).src = imgUrl;
            const parent = thumbElement.parentElement;
            const thumbs = parent.getElementsByClassName('thumb-img');
            for(let t of thumbs) { t.classList.remove('active'); }
            thumbElement.classList.add('active');
        }

        function addToCart(index) {
            const product = products[index];
            const selectedSize = document.getElementById(`size-${index}`).value;
            
            cart.push({ ...product, size: selectedSize, mainImg: document.getElementById(`main-img-${index}`).src });
            updateCart();
            toggleCart();
        }

        function updateCart() {
            document.getElementById('cart-count').innerText = cart.length;
            const itemsContainer = document.getElementById('cart-items');
            
            if(cart.length === 0) {
                itemsContainer.innerHTML = "<p style='color: var(--text-gray); margin-top:20px;'>Your cart is empty.</p>";
            } else {
                itemsContainer.innerHTML = cart.map((item, index) => `
                    <div class="cart-item">
                        <div>
                            <strong>${item.name}</strong><br>
                            <small style="color: var(--text-gray);">${item.size}</small><br>
                            <small style="color: var(--accent-red);">LKR ${Number(item.price).toLocaleString()}</small>
                        </div>
                        <span style="color:#ef4444; font-weight:bold; cursor:pointer; font-size:20px;" onclick="removeFromCart(${index})">&times;</span>
                    </div>
                `).join('');
            }
            
            const total = cart.reduce((sum, item) => sum + Number(item.price), 0);
            document.getElementById('cart-total').innerText = total.toLocaleString();
        }

        function removeFromCart(index) {
            cart.splice(index, 1);
            updateCart();
        }

        function toggleCart() {
            document.getElementById('cart-modal').classList.toggle('active');
        }

        /* --- Admin Panel Functions --- */

            if (!isAdminLoggedIn) {
                let password = prompt("Enter Admin Password:");
                if (password === "Dinzzo@2007") {
                    isAdminLoggedIn = true;
                    document.getElementById('admin-panel').classList.add('active');
                    renderProducts();
                    alert("Admin Access Granted!");
                } else {

                }
            } else {

            }
        }

        function addNewProduct() {
            const name = document.getElementById('new-name').value;
            const price = parseFloat(document.getElementById('new-price').value);
            const fileInput = document.getElementById('new-img-files');

            if (!name || !price || fileInput.files.length === 0) {
                alert("Please fill all details and select at least one image file!");
                return;
            }

            if (fileInput.files.length > 10) {
                alert("You can only upload a maximum of 10 images per product!");
                return;
            }

            const files = Array.from(fileInput.files);
            const imagePromises = files.map(file => {
                return new Promise((resolve) => {
                    const reader = new FileReader();
                    reader.onload = e => resolve(e.target.result);
                    reader.readAsDataURL(file);
                });
            });

            Promise.all(imagePromises).then(base64Images => {
                database.ref('products').push({
                    name: name,
                    price: price,
                    images: base64Images
                }).then(() => {
                    document.getElementById('new-name').value = '';
                    document.getElementById('new-price').value = '';
                    fileInput.value = '';
                    alert("New product with multiple images added successfully!");
                }).catch(error => {
                    alert("Error adding product: " + error.message);
                });
            });
        }

        function deleteProduct(firebaseId) {
            if (confirm("Are you sure you want to delete this product?")) {
                database.ref('products/' + firebaseId).remove()
                .then(() => alert("Product deleted successfully!"))
                .catch(error => alert("Error deleting product: " + error.message));
            }
        }

        function sendWhatsAppOrder() {
            if (cart.length === 0) { alert('Your cart is empty!'); return; }
            
            let phone = "94740597963"; 
            
            let message = "Hi DBD SPORTS! I would like to place an order:%0A%0A";
            cart.forEach((item, i) => {
                message += `${i + 1}. ${item.name} - ${item.size} (LKR ${Number(item.price).toLocaleString()})%0A`;
            });
            const total = cart.reduce((sum, item) => sum + Number(item.price), 0);
            message += `%0A*Total Amount: LKR ${total.toLocaleString()}*`;
            
            window.open(`https://wa.me/${phone}?text=${message}`, '_blank');
        }
